    finally:
        # Only the group created above is signalled. This stops compute, and
        # preserves every source file, checkpoint, log and partial result.
        try:
            if process is not None:
                if process.poll() is None:
                    try:
                        os.killpg(process.pid, signal.SIGTERM)
                    except ProcessLookupError:
                        pass
                    try:
                        process.wait(timeout=30)
                    except subprocess.TimeoutExpired:
                        try:
                            os.killpg(process.pid, signal.SIGKILL)
                        except ProcessLookupError:
                            pass
                        process.wait(timeout=30)
                try:
                    os.killpg(process.pid, signal.SIGKILL)
                except ProcessLookupError:
                    pass
        except BaseException as cleanup_exc:
            cleanup_error = repr(cleanup_exc)
        receipt = dict(status='EXIT_ZERO' if process is not None and process.returncode == 0
                       and error is None and cleanup_error is None else 'FAILED_OR_INTERRUPTED',
                       return_code=None if process is None else process.returncode,
                       timed_out=timed_out, error=error, cleanup_error=cleanup_error,
                       elapsed_seconds=round(time.monotonic() - started, 3),
                       source_sha256=STRUCTURE_UNIFORM_SHA256,
                       final_test_used=False, official_score=None)
        with (source_dir / 'NOTEBOOK_DRIVER_RECEIPT.json').open('x') as stream:
            json.dump(receipt, stream, indent=2)
        print('STRUCTURE_UNIFORM_DRIVER_RECEIPT', json.dumps(receipt), flush=True)
    assert process.returncode == 0 and cleanup_error is None, 'Driver failed; retain all logs and partial outputs'
